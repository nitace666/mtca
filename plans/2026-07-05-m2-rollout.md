```python
"""benchmarks/accuracy_report.py — T30 召回准确率量化。
跑 test_set.json，对每条 query 跑 recall(query, top_k=3)，
统计 hit_rate@1 / @3 / MRR，输出 markdown。
"""
import json
from pathlib import Path
from src.recall.recall_engine import recall


def measure(path: Path, test_set: list[dict]) -> dict:
    hits_at_1 = hits_at_3 = 0
    reciprocal_ranks = []
    for case in test_set:
        results = recall(query=case["query"], top_k=3, path=path)
        seg_ids = [r["segment_id"] for r in results]
        if case["expected_top1"] in seg_ids:
            rank = seg_ids.index(case["expected_top1"]) + 1
            reciprocal_ranks.append(1.0 / rank)
            if rank == 1:
                hits_at_1 += 1
            if rank <= 3:
                hits_at_3 += 1
        else:
            reciprocal_ranks.append(0.0)
    n = len(test_set)
    return {
        "hit_rate@1": hits_at_1 / n,
        "hit_rate@3": hits_at_3 / n,
        "MRR": sum(reversal_ranks := reciprocal_ranks) / n if n else 0,
        "n": n,
    }


def main():
    test_set = json.loads(Path("benchmarks/test_set.json").read_text(encoding="utf-8"))
    # 取默认 db
    metrics = measure(Path("C:/Users/Administrator/.mtca/mtca.db"), test_set)
    md = f"""# MTCA 召回准确率报告（M2）

| 指标 | 值 |
|---|---|
| hit_rate@1 | {metrics['hit_rate@1']:.3f} |
| hit_rate@3 | {metrics['hit_rate@3']:.3f} |
| MRR | {metrics['MRR']:.3f} |
| 测试集大小 | {metrics['n']} |

生成时间：见 git log
"""
    Path("benchmarks/M2_accuracy_report.md").write_text(md, encoding="utf-8")
    print(md)


if __name__ == "__main__":
    main()
```

```python
"""src/store/backup.py — T33 自动备份。"""
import shutil
import time
from datetime import datetime, timedelta
from pathlib import Path
from typing import Optional, Union


def backup_now(
    path: Optional[Union[Path, str]] = None,
    backup_dir: Optional[Path] = None,
    keep_days: int = 7,
) -> Path:
    """复制 db 到 backup_dir/mtca-YYYYMMDD-HHMMSS.db，清理 > keep_days 的老备份。
    
    返回本次备份文件路径。
    """
    from src.store.sqlite import MTCA_DB_PATH
    src = Path(path) if path else MTCA_DB_PATH
    bdir = backup_dir or src.parent / "backups"
    bdir.mkdir(parents=True, exist_ok=True)
    
    ts = datetime.now().strftime("%Y%m%d-%H%M%S")
    dest = bdir / f"mtca-{ts}.db"
    shutil.copy2(src, dest)
    
    # 清理过期
    cutoff = time.time() - keep_days * 86400
    for old in bdir.glob("mtca-*.db"):
        if old.stat().st_mtime < cutoff:
            old.unlink()
    
    return dest
```

```python
import time
from pathlib import Path

    result = backup_now(path=db, backup_dir=bdir)

def test_backup_cleans_old(tmp_path):
    db = tmp_path / "mtca.db"
```

```python
"""src/adapters/mcp_server.py — T31 MCP Server（stdio）。"""
import asyncio
import json
import sys
from pathlib import Path
from typing import Any

# 4 个 tool，复用现有逻辑
from src.recall.recall_engine import recall, recall_with_fallback
from src.compress.scoring import mark_important
from src.fog.fog_engine import fog_segment


TOOLS = [
    {
        "name": "recall_segments",
        "description": "按 query 召回段落（返回 L0 原文）",
        "params": {
            "query": str,
            "top_k": int (default 5),
            "time_window": [int, int] (optional),
        },
        "handler": lambda **p: recall(**p),
    },
    {
        "name": "mark_important",
        "description": "/重要 命令直达 L0",
        "params": {"segment_id": str},
        "handler": lambda **p: {"status": "ok", "rows": mark_important(**p)},
    },
    {
        "name": "fog_segment",
        "description": "/雾化（需 anchor）",
        "params": {"segment_id": str, "anchor": str},
        "handler": lambda **p: {"status": "ok", "ok": fog_segment(**p, called_by="user")},
    },
    {
        "name": "get_recent_sessions",
        "description": "列最近会话（fallback 用）",
        "params": {"limit": int (default 10)},
        "handler": lambda **p: recall_with_fallback(query="", top_k=p.get("limit", 10)),
    },
]


def handle_request(req: dict[str, Any]) -> dict[str, Any]:
    """处理单条 JSON-RPC 风格请求。"""
    tool_name = req.get("tool")
    params = req.get("params", {})
    tool = next((t for t in TOOLS if t["name"] == tool_name), None)
    if not tool:
        return {"error": f"unknown tool: {tool_name}"}
    try:
        return {"result": tool["handler"](**params)}
    except Exception as e:
        return {"error": str(e)}


def main() -> None:
    """stdio loop：读 stdin JSON 行，回写 stdout JSON 行。"""
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        req = json.loads(line)
        resp = handle_request(req)
        sys.stdout.write(json.dumps(resp, ensure_ascii=False) + "\n")
        sys.stdout.flush()


if __name__ == "__main__":
    main()
```
